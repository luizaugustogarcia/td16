cmake_minimum_required(VERSION 3.24)
project(td_experiments_cuda LANGUAGES CUDA)
find_package(CUDAToolkit REQUIRED)
set(TDP_CUDA_ARCHITECTURES "" CACHE STRING "Compute capabilities, e.g. 89;120; empty detects visible GPUs")
if(NOT TDP_CUDA_ARCHITECTURES)
  execute_process(COMMAND nvidia-smi --query-gpu=compute_cap --format=csv,noheader OUTPUT_VARIABLE caps OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  string(REPLACE "." "" caps "${caps}")
  string(REPLACE "\n" ";" TDP_CUDA_ARCHITECTURES "${caps}")
endif()
list(REMOVE_DUPLICATES TDP_CUDA_ARCHITECTURES)
set(resources "${CMAKE_CURRENT_SOURCE_DIR}/../main/resources/cuda")
set(emitter "${CMAKE_CURRENT_SOURCE_DIR}/../../../emitter/src/native/backtracking/sorting_search.cu")
file(READ "${emitter}" emitter_source)
string(FIND "${emitter_source}" "    __device__ __forceinline__ int count_even_cycles" begin)
string(FIND "${emitter_source}" "    // ─── Pruning logic" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Cannot locate emitter cycle audit functions")
endif()
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${emitter_source}" ${begin} ${length} cycle_functions)
file(READ "${resources}/cycle-delta-audit.cu" audit)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/emitter-delta-audit.cu" "#include <cstdint>\n#include <climits>\n${cycle_functions}${audit}")
file(READ "${resources}/streaming-count.cu" streaming_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/streaming-count.cu" "#include <cstdint>\n#include <climits>\n${cycle_functions}${streaming_source}")
set(outputs)
foreach(arch IN LISTS TDP_CUDA_ARCHITECTURES)
  if(NOT arch MATCHES "^[0-9]+$")
    message(FATAL_ERROR "Expected numeric CUDA architecture, got '${arch}'")
  endif()
  foreach(kernel diametral-filter streaming-count-soa streaming-count-aos emitter-delta-audit)
    set(definition)
    if(kernel MATCHES "streaming-count")
      set(source "${CMAKE_CURRENT_BINARY_DIR}/streaming-count.cu")
      if(kernel MATCHES "soa$")
        set(definition -DSOA_STACK)
      else()
        set(definition -DAOS_STACK)
      endif()
    elseif(kernel STREQUAL "emitter-delta-audit")
      set(source "${CMAKE_CURRENT_BINARY_DIR}/emitter-delta-audit.cu")
    else()
      set(source "${resources}/${kernel}.cu")
    endif()
    set(output "${CMAKE_CURRENT_BINARY_DIR}/${kernel}-compute_${arch}.ptx")
    add_custom_command(OUTPUT "${output}" COMMAND "${CMAKE_CUDA_COMPILER}" --ptx --std=c++11 --generate-line-info --gpu-architecture=compute_${arch} ${definition} "${source}" -o "${output}" DEPENDS "${source}" VERBATIM)
    list(APPEND outputs "${output}")
  endforeach()
endforeach()
add_custom_target(cuda_kernels ALL DEPENDS ${outputs})
